// Show ONP navbar on scroll const myLink = document.getElementById('onpNav'); // Define the pixel threshold to trigger the style change const scrollThreshold = 40; // Listen for the window scroll event window.addEventListener('scroll', () => { // Check the current vertical scroll position if (window.scrollY >= scrollThreshold) { myLink.classList.remove('scrolled-past'); } else { myLink.classList.add('scrolled-past'); } }); .scrolled-past { display: none !important; transition: color 0.3s ease !important; }
